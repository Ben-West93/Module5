# --- auth (L8) ---
python-jose[cryptography]>=3.3      # JWT encode/decode
passlib[bcrypt]>=1.7.4              # password hashing

# passlib 1.7.4 reads bcrypt.__about__.__version__, which bcrypt 4.1 removed.
# With a newer bcrypt, passlib mis-detects the backend and rejects every
# password with a bogus "longer than 72 bytes" error — even a short one.
# Pinning below that break keeps bcrypt working. app/utils/security.py also
# falls back to pbkdf2_sha256 if this pin is ignored, so the app still starts.
bcrypt==4.0.1
